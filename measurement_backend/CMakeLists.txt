cmake_minimum_required(VERSION 3.16)
project(sm9_revision_measurement C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(vendor/gmssl-src EXCLUDE_FROM_ALL)
add_library(siap_crypto SHARED native/crypto.c native/sm2.c native/miller.c)
target_include_directories(siap_crypto PRIVATE vendor/gmssl-src/include)
target_link_libraries(siap_crypto PRIVATE gmssl)
target_compile_options(siap_crypto PRIVATE -O3)
set_target_properties(siap_crypto PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_library(kernel SHARED experiments/kernel_screen_20261021/kernel.c native/miller.c)
target_include_directories(kernel PRIVATE vendor/gmssl-src/include)
target_link_libraries(kernel PRIVATE gmssl)
target_compile_options(kernel PRIVATE -O3)
set_target_properties(kernel PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/experiments/kernel_screen_20261021")
add_library(bridge SHARED experiments/minscreen_20261020/bridge.c native/miller.c)
target_include_directories(bridge PRIVATE vendor/gmssl-src/include)
target_link_libraries(bridge PRIVATE gmssl)
target_compile_options(bridge PRIVATE -O3)
set_target_properties(bridge PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/experiments/minscreen_20261020")
